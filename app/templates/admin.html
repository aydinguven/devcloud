cd devcloud-worker
sudo install -d -m 0755 /etc/devcloud
sudo install -m 0600 deploy/worker.env.example /etc/devcloud/worker.env
sudo vi /etc/devcloud/worker.env
sudo bash deploy/deploy_worker_offline.sh
